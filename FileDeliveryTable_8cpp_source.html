31 // MBMS / FLUTE FDT namespace URIs (TS 26.346 cl. 7.2.10 + per-release
32 // XSD overlays). Senders may bind these URIs to any prefix they like
33 // (`mbms2007:`, `cc:`, `n1:`, …) so the parser MUST resolve qualified
34 // names against the document's xmlns declarations rather than
35 // pattern-matching on the literal prefix.
36 constexpr std::string_view kNsMbms2007 = "urn:3GPP:metadata:2007:MBMS:FLUTE:FDT";
37 constexpr std::string_view kNsMbms2008 = "urn:3GPP:metadata:2008:MBMS:FLUTE:FDT_ext";
38 constexpr std::string_view kNsMbms2009 = "urn:3GPP:metadata:2009:MBMS:FLUTE:FDT_ext";
39 constexpr std::string_view kNsMbms2012 = "urn:3GPP:metadata:2012:MBMS:FLUTE:FDT";
40 constexpr std::string_view kNsMbms2025 = "urn:3GPP:metadata:2025:MBMS:FLUTE:FDT";
41 constexpr std::string_view kNsSchemaVersion =
42  "urn:3GPP:metadata:2009:MBMS:schemaVersion";

44 // Map xmlns prefix → namespace URI as declared on the FDT-Instance
45 // root. Real XML allows xmlns to be redefined in nested elements; we
46 // don't (MBMS FDTs don't do that in practice), and keep this scoped to
47 // the document root for simplicity. The empty key represents the
48 // no-namespace bucket, where unqualified attribute / element names
49 // like `Expires`, `File`, `TOI` live.
50 using NsMap = std::unordered_map<std::string, std::string>;
51 
52 NsMap ReadXmlnsDeclarations(const tinyxml2::XMLElement* e) {
53  NsMap m;
54  m[std::string{}] = std::string{}; // unprefixed attrs are in no-namespace
55  for (const tinyxml2::XMLAttribute* a = e->FirstAttribute();
56  a != nullptr; a = a->Next()) {
57  const std::string name = a->Name() != nullptr ? a->Name() : "";
58  const std::string val = a->Value() != nullptr ? a->Value() : "";
59  if (name.rfind("xmlns:", 0) == 0) {
60  m[name.substr(6)] = val;
61  }
62  }
63  return m;
64 }

66 // Split "prefix:local" into (prefix, local). If no colon, prefix == "".
67 std::pair<std::string_view, std::string_view> SplitQName(std::string_view qn) {
68  auto pos = qn.find(':');
69  if (pos == std::string_view::npos) return {std::string_view{}, qn};
70  return {qn.substr(0, pos), qn.substr(pos + 1)};
71 }

73 // True when `qn` (a qualified name from the document) resolves to
74 // (target_uri, target_local) under the supplied prefix→URI map.
75 bool QNameMatches(std::string_view qn, const NsMap& ns,
76  std::string_view target_uri, std::string_view target_local) {
77  auto [prefix, local] = SplitQName(qn);
78  if (local != target_local) return false;
79  auto it = ns.find(std::string(prefix));
80  if (it == ns.end()) return false;
81  return it->second == target_uri;
82 }

84 // Find the first child of `parent` whose qualified name resolves to
85 // (uri, local-name). Returns nullptr if no match.
86 tinyxml2::XMLElement*
87 FindChildByNs(tinyxml2::XMLElement* parent, const NsMap& ns,
88  std::string_view target_uri, std::string_view target_local) {
89  for (auto* c = parent->FirstChildElement(); c != nullptr;
90  c = c->NextSiblingElement()) {
91  if (c->Name() != nullptr &&
92  QNameMatches(c->Name(), ns, target_uri, target_local)) {
93  return c;
94  }
95  }
96  return nullptr;
97 }

99 // Look up an attribute on `e` by namespace URI + local name. Skips the
100 // xmlns:* declarations themselves.
101 const char* FindAttrByNs(const tinyxml2::XMLElement* e, const NsMap& ns,
102  std::string_view target_uri,
103  std::string_view target_local) {
104  for (const auto* a = e->FirstAttribute(); a != nullptr; a = a->Next()) {
105  const std::string n = a->Name() != nullptr ? a->Name() : "";
106  if (n == "xmlns" || n.rfind("xmlns:", 0) == 0) continue;
107  if (QNameMatches(n, ns, target_uri, target_local)) {
108  return a->Value();
109  }
110  }
111  return nullptr;
112 }

116 // RFC 6726 §3.3 instance-ID space is 20 bits (the EXT_FDT field
117 // reserves 20 bits). RFC 1982 serial-number arithmetic over a 2^20
118 // circle: half the space (2^19) is "ahead", the other half is
119 // "behind". We compute the forward distance modulo 2^20 and call
120 // the candidate "newer" iff the forward distance is in (0, 2^19).
121 bool LibFlute::FileDeliveryTable::IsNewerInstanceId(uint32_t candidate,
122  uint32_t current) {
123  constexpr uint32_t kModulus = 1u << 20;
124  constexpr uint32_t kHalf = 1u << 19;
125  const uint32_t cand = candidate & (kModulus - 1u);
126  const uint32_t cur = current & (kModulus - 1u);
127  const uint32_t fwd = (cand - cur) & (kModulus - 1u);
128  return fwd != 0u && fwd < kHalf;
129 }

360  // TS 26.346 cl. 7.2.10.2 (Rel-7 mbms2007): <Cache-Control> is an
361  // <xs:choice> of three alternatives — exactly one of <Expires>,
362  // <no-cache> or <max-stale>. Documents carrying more than one are
363  // XSD-invalid; reject them rather than picking arbitrarily.
364  uint64_t expires = 0;
365  FileDeliveryTable::CacheControl cache_control =
366  FileDeliveryTable::CacheControl::Expires;
367  if (auto* cc = FindChildByNs(file, ns, kNsMbms2007, "Cache-Control");
368  cc != nullptr) {
369  int child_count = 0;
370  bool have_expires = false;
371  bool have_nocache = false;
372  bool have_maxstale = false;
373  for (auto* c = cc->FirstChildElement(); c != nullptr;
374  c = c->NextSiblingElement()) {
375  ++child_count;
376  if (c->Name() == nullptr) continue;
377  const std::string_view qn = c->Name();
378  if (QNameMatches(qn, ns, kNsMbms2007, "Expires")) {
379  have_expires = true;
380  if (const char* t = c->GetText(); t != nullptr) {
381  expires = strtoull(t, nullptr, 0);
382  }
383  } else if (QNameMatches(qn, ns, kNsMbms2007, "no-cache")) {
384  have_nocache = true;
385  } else if (QNameMatches(qn, ns, kNsMbms2007, "max-stale")) {
386  have_maxstale = true;
387  }
388  }
389  if (child_count > 1) {
390  throw std::runtime_error(
391  "mbms2007:Cache-Control violates xs:choice — multiple children present");
392  }
393  if (have_nocache) cache_control = FileDeliveryTable::CacheControl::NoCache;
394  else if (have_maxstale) cache_control = FileDeliveryTable::CacheControl::MaxStale;
395  else if (have_expires) cache_control = FileDeliveryTable::CacheControl::Expires;
396  }

495 auto LibFlute::FileDeliveryTable::to_string() const -> std::string {
496  tinyxml2::XMLDocument doc;
497  doc.InsertFirstChild( doc.NewDeclaration() );
498  auto* root = doc.NewElement("FDT-Instance");
499  root->SetAttribute("Expires", std::to_string(_expires).c_str());
500  root->SetAttribute("FEC-OTI-FEC-Encoding-ID", (unsigned)_global_fec_oti.encoding_id);
501  root->SetAttribute("FEC-OTI-Maximum-Source-Block-Length", (unsigned)_global_fec_oti.max_source_block_length);
502  root->SetAttribute("FEC-OTI-Encoding-Symbol-Length", (unsigned)_global_fec_oti.encoding_symbol_length);
503  // TS 26.346 §7.2.10.1: emit Instance-ID / Max-Number-of-Encoding-Symbols
504  // / Scheme-Specific-Info on the FDT-Instance root only if the encoder's
505  // FDT-Instance defaults set them. Sentinel (0 / empty) ⇒ omit, matching
506  // the spec's "absent attribute = no default applies" semantics.
507  if (_global_fec_oti.instance_id != 0) {
508  root->SetAttribute("FEC-OTI-FEC-Instance-ID",
509  (unsigned)_global_fec_oti.instance_id);
510  }
511  if (_global_fec_oti.max_number_of_encoding_symbols != 0) {
512  root->SetAttribute("FEC-OTI-Max-Number-of-Encoding-Symbols",
513  (unsigned)_global_fec_oti.max_number_of_encoding_symbols);
514  }
515  if (!_global_fec_oti.scheme_specific_info.empty()) {
516  root->SetAttribute("FEC-OTI-Scheme-Specific-Info",
517  _global_fec_oti.scheme_specific_info.c_str());
518  }
519  // TS 26.346 cl. 7.2.10 + per-release XSDs: declare all the MBMS
520  // namespaces this implementation may emit, plus the schema-version
521  // namespace. Receivers expect these prefixes when reading the
522  // qualified attributes/elements below.
523  root->SetAttribute("xmlns:mbms2007", "urn:3GPP:metadata:2007:MBMS:FLUTE:FDT");
