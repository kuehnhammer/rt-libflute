96  spdlog::set_pattern("[%H:%M:%S.%f] [%^%l%$] %v");

98  // Mmap each input file. The Encoder takes a non-owning pointer; the
99  // mmap stays alive until completion-callback frees it.
100  struct Mapped {
101  std::string path;
102  char* buffer = nullptr;
103  std::size_t length = 0;
104  std::uint16_t toi = 0;
105  };
106  std::vector<Mapped> files;
107  for (int j = 0; args.files && args.files[j]; ++j) {
108  int fd = open(args.files[j], O_RDONLY);
109  if (fd < 0) {
110  spdlog::error("open({}): {}", args.files[j], std::strerror(errno));
111  continue;
112  }
113  struct stat st;
114  if (fstat(fd, &st) < 0 || st.st_size <= 0) {
115  close(fd);
116  continue;
117  }
118  void* p = mmap(nullptr, st.st_size, PROT_READ, MAP_PRIVATE, fd, 0);
119  close(fd);
120  if (p == MAP_FAILED) {
121  spdlog::error("mmap({}): {}", args.files[j], std::strerror(errno));
122  continue;
123  }
124  files.push_back({args.files[j], static_cast<char*>(p),
125  static_cast<std::size_t>(st.st_size), 0});
126  }
127  if (files.empty()) {
128  spdlog::error("no usable input files");
129  return 1;
130  }
131 
132  // Open a plain UDP socket. Multicast TTL 8, loopback enabled so a
133  // local receiver on the same host can pick the packets up.
134  int sock = socket(AF_INET, SOCK_DGRAM, 0);
135  if (sock < 0) {
136  spdlog::error("socket: {}", std::strerror(errno));
137  return 1;
138  }
139  unsigned char ttl = 8;
140  setsockopt(sock, IPPROTO_IP, IP_MULTICAST_TTL, &ttl, sizeof(ttl));
141  unsigned char loop = 1;
142  setsockopt(sock, IPPROTO_IP, IP_MULTICAST_LOOP, &loop, sizeof(loop));
143 
144  sockaddr_in dst{};
145  dst.sin_family = AF_INET;
146  dst.sin_port = htons(args.mcast_port);
147  if (inet_pton(AF_INET, args.mcast_target, &dst.sin_addr) != 1) {
148  spdlog::error("invalid target address: {}", args.mcast_target);
149  close(sock);
150  return 1;
151  }
152 
153  // Encoder: PacketCallback dispatches via sendto. Returns true on
154  // successful dispatch so the encoder marks symbols transmitted.
155  LibFlute::Encoder encoder(
156  args.tsi, args.mtu, args.rate_limit_kbps,
157  [sock, &dst](std::span<const std::uint8_t> bytes) -> bool {
158  ssize_t n = sendto(sock, bytes.data(), bytes.size(), 0,
159  reinterpret_cast<const sockaddr*>(&dst),
160  sizeof(dst));
161  if (n < 0) {
162  spdlog::warn("sendto: {}", std::strerror(errno));
163  return false;
164  }
165  return true;
166  });
167 
168  encoder.register_completion_callback([&](std::uint32_t toi) {
169  for (auto& f : files) {
170  if (f.toi == toi) {
171  spdlog::info("transmitted {} (TOI {})", f.path, toi);
172  munmap(f.buffer, f.length);
173  f.buffer = nullptr;
174  }
175  }
176  });
177 
178  for (auto& f : files) {
179  f.toi = encoder.send(
180  f.path, "application/octet-stream",
181  LibFlute::Encoder::seconds_since_epoch() + 60,
182  f.buffer, f.length,
183  static_cast<LibFlute::FecScheme>(args.fec),
184  /*copy_buffer=*/false);
185  if (f.toi > 0) {
186  spdlog::info("queued {} ({} bytes, TOI {})", f.path, f.length, f.toi);
187  }
188  }
189 
190  // Pump packets until everything's been transmitted. Honour the
191  // encoder's rate-limit deadline by sleeping until the next due time.
192  while (encoder.has_pending()) {
193  if (!encoder.send_next_packet()) {
194  auto due = encoder.next_send_due();
195  if (due) {
196  auto now = std::chrono::steady_clock::now();
197  if (*due > now) std::this_thread::sleep_until(*due);
198  } else {
199  break;
200  }
201  }
202  }
203  // Drain any FDT-only packets the encoder might still have queued.
204  encoder.flush();
205 
206  close(sock);
207  return 0;
208 }
