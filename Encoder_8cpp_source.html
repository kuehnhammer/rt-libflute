26 // IP+UDP+LCT-base+SBN/ESI overhead before the encoding-symbol payload.

65 Encoder::OverheadEstimate
66 Encoder::EstimateOverhead(const Encoder::OverheadParameters& p) {
67  // Header sizes for the per-packet overhead. The FLUTE encoder in
68  // this build emits a 12-byte LCT header on file packets (LCT base
69  // 4 + CCI 4 + half-word TSI 2 + half-word TOI 2; toi_flag=0,
70  // half_word_flag=1) and a 32-byte LCT header on FDT packets
71  // (TOI=0 ⇒ EXT_FDT 4 + EXT_FTI 16 added). For both Raptor and
72  // CompactNoCode the FEC payload ID is 4 bytes (SBN+ESI). UDP and
73  // IPv4/IPv6 are RFC standard.
74  constexpr unsigned int kUdpHdr = 8;
75  constexpr unsigned int kIpv4Hdr = 20;
76  constexpr unsigned int kIpv6Hdr = 40;
77  constexpr unsigned int kLctHdrFile = 12;
78  constexpr unsigned int kLctHdrFdt = 32;
79  constexpr unsigned int kFecPayloadId = 4;
80 
81  const unsigned int ip_hdr =
82  p.ipv6 ? kIpv6Hdr : kIpv4Hdr;
83  const unsigned int file_packet_overhead =
84  ip_hdr + kUdpHdr + kLctHdrFile + kFecPayloadId;
85  const unsigned int fdt_packet_overhead =
86  ip_hdr + kUdpHdr + kLctHdrFdt + kFecPayloadId;
87 
88  OverheadEstimate r{};
89  r.payload_bps = p.payload_bps;
90  if (p.mtu <= file_packet_overhead) {
91  // MTU too small to carry any payload; can only return the input.
92  r.total_bps = p.payload_bps;
93  return r;
94  }
95 
96  const unsigned int file_payload_per_packet = p.mtu - file_packet_overhead;
97 
98  // FEC repair-symbol contribution. CompactNoCode has no repair
99  // path; Raptor scales source bps by `fec_redundancy` to get the
100  // per-second repair byte rate.
101  double redundancy = p.fec_redundancy;
102  if (p.fec_scheme == FecScheme::CompactNoCode || redundancy < 0.0) {
103  redundancy = 0.0;
104  }
105  r.fec_repair_bps =
106  static_cast<std::uint64_t>(static_cast<double>(p.payload_bps) * redundancy);
107 
108  // Per-packet header overhead amortised across the packet rate
109  // implied by (payload + repair) / file_payload_per_packet.
110  // Algebraically equivalent to:
111  // header_bps = (payload_bps + fec_repair_bps)
112  // * (file_packet_overhead / file_payload_per_packet)
113  // expressed via the packet-rate intermediate so the rounding
114  // matches the formula the application has been computing.
115  const double effective_payload_bps =
116  static_cast<double>(p.payload_bps + r.fec_repair_bps);
117  const double pkt_rate =
118  effective_payload_bps / (8.0 * static_cast<double>(file_payload_per_packet));
119  r.packet_header_bps = static_cast<std::uint64_t>(
120  pkt_rate * 8.0 * static_cast<double>(file_packet_overhead));
121 
122  // FDT broadcast amortised over `fdt_period_seconds`. The FDT body
123  // is `fdt_size_bytes` of XML, fragmented into ⌈body / payload⌉
124  // packets each carrying its own header overhead.
125  if (p.fdt_period_seconds > 0 && p.mtu > fdt_packet_overhead) {
126  const unsigned int fdt_payload_per_packet = p.mtu - fdt_packet_overhead;
127  const std::uint64_t fdt_packets_per_period =
128  (static_cast<std::uint64_t>(p.fdt_size_bytes) + fdt_payload_per_packet - 1)
129  / fdt_payload_per_packet;
130  const std::uint64_t fdt_bytes_per_period =
131  fdt_packets_per_period
132  * static_cast<std::uint64_t>(fdt_packet_overhead + fdt_payload_per_packet);
133  r.fdt_bps = (fdt_bytes_per_period * 8ULL) / p.fdt_period_seconds;
134  }
135 
136  r.total_bps = r.payload_bps + r.fec_repair_bps
137  + r.packet_header_bps + r.fdt_bps;
138  return r;
139 }

151 std::uint16_t Encoder::send(std::string content_location,
152  std::string content_type,
153  std::uint64_t expires_ntp_seconds,
154  char* data, std::size_t length,
155  FileTransmissionConfig fec_config,
156  bool copy_buffer) {
157  const std::lock_guard<std::mutex> lock(_mutex);
158  if (!data || length == 0) {
159  return 0;
160  }
161 
162  const std::uint16_t toi = _next_toi;
163  _next_toi = static_cast<std::uint16_t>(_next_toi + 1);
164  if (_next_toi == 0) {
165  // 0 is reserved for the FDT (RFC 6726 §3.1); skip on wrap.
166  _next_toi = 1;
167  }
168 
169  // The caller supplies only what TS 26.346 SDP exposes (§7.3.2.8 +
170  // §7.3.2.11): scheme + instance-id + redundancy-level. Everything
171  // else in the FEC OTI (T, K, Z, N, Al, max-encoding-symbols) is
172  // libflute's job — derived by the FEC transformer from F, mtu, and
173  // the partitioner's algorithm, then written into the FDT for the
174  // receiver. The encoder's `_fec_oti` provides the FDT-Instance
175  // defaults for non-FEC fields (T from MTU, etc.).
176  FecOti file_oti = _fec_oti;
177  file_oti.encoding_id = fec_config.scheme;
178  file_oti.instance_id = fec_config.fec_instance_id;
179 
180  std::shared_ptr<File> file;
181  try {
182  file = std::make_shared<File>(toi, file_oti, std::move(content_location),
183  std::move(content_type),
184  expires_ntp_seconds, data, length,
185  copy_buffer,
186  fec_config.fec_redundancy_level,
187  _fec_worker_threads,
188  fec_config.sub_block_size_target,
189  fec_config.sub_symbol_size_min_multiplier);
190  } catch (const std::exception& ex) {
191  spdlog::error("Encoder::send: failed to create File: {}", ex.what());
192  return 0;
193  }
194 
195  // Carry the Rel-11 FEC-Redundancy-Level alongside the FEC OTI on the
196  // FileEntry so the FDT serialiser emits the mbms2012 attribute.
197  file->meta().fec_redundancy_level = fec_config.fec_redundancy_level;
198 
199  _fdt->add(file->meta());
200  _files.insert({toi, file});
201  queue_fdt_locked();
202  _stats.files_queued.fetch_add(1, std::memory_order_relaxed);
203  return toi;
204 }

219 void Encoder::queue_fdt_locked() {
220  // Empty-FDT defence. At least one commercial MBMS middleware
221  // (Qualcomm) crashes hard on receipt of an FDT-Instance that
222  // lists zero <File> entries; centralising the skip here covers
223  // both the carousel path (requeue_fdt → caller-driven) and the
224  // event-driven on-completion path (file_transmitted_locked →
225  // FDT becomes empty when the last queued file finishes). One
226  // check at the top of queue_fdt_locked is the single point of
227  // truth.
228  if (_fdt->file_entries().empty()) {
229  spdlog::trace("Encoder::queue_fdt: FDT has zero file entries — skipping emit");
230  return;
231  }
232 
233  // Refresh the FDT XML and queue it as the special TOI=0 File. Any
234  // previous incomplete FDT is discarded — receivers handle in-flight
235  // FDT supersession via the round-4 instance-ID monotonicity logic.
236  const std::uint64_t expires_ntp =
237  seconds_since_epoch() + _fdt_expires_window_seconds;
238  _fdt->set_expires(expires_ntp);
239  auto fdt_xml = _fdt->to_string();
240 
241  FecOti fdt_oti = _fec_oti;
242  fdt_oti.encoding_id = FecScheme::CompactNoCode; // FDT is never FEC-encoded
243 
244  try {
245  auto fdt_file = std::make_shared<File>(
246  /*toi*/ 0, fdt_oti, std::string{}, std::string{},
247  expires_ntp,
248  fdt_xml.data(), fdt_xml.size(),
249  /*copy_data*/ true);
250  fdt_file->set_fdt_instance_id(_fdt->instance_id());
251  _files.insert_or_assign(0, std::move(fdt_file));
252  } catch (const std::exception& ex) {
253  spdlog::error("Encoder::queue_fdt: {}", ex.what());
254  }
255 }

262 void Encoder::file_transmitted_locked(std::uint32_t toi) {
263  if (toi == 0) {

265  // the map (complete) until the next queue_fdt_locked() replaces it.

294 bool Encoder::send_next_packet() {
295  std::shared_ptr<File> completed_file;
296  std::uint32_t completed_toi = 0;
297  CompletionCallback dispatch_cb;
298 
299  {
300  const std::lock_guard<std::mutex> lock(_mutex);
301  if (!_running) return false;
302  if (Clock::now() < _next_send_due) return false;
303 
304  File* sending_file = nullptr;
305  for (auto& [toi, file] : _files) {
306  if (file && !file->complete()) {
307  sending_file = file.get();
308  break;
309  }
310  }
311  if (sending_file == nullptr) return false;
312 
313  auto symbols = sending_file->get_next_symbols(_max_payload);
314  if (symbols.empty()) {
315  // Pathological: file isn't complete but no symbols left to queue
316  // (every symbol is already queued or every block is somehow
317  // complete-but-not-checked). Treat as nothing to send right now.
318  return false;
319  }
320 
321  AlcPacket packet(
322  static_cast<std::uint16_t>(_tsi),
323  static_cast<std::uint16_t>(sending_file->meta().toi),
324  sending_file->meta().fec_oti, symbols, _max_payload,
325  sending_file->fdt_instance_id(),
326  std::span<std::uint8_t>(_packet_scratch.data(),
327  _packet_scratch.size()));
328 
329  bool dispatched = false;
330  if (_packet_cb) {
331  dispatched = _packet_cb({_packet_scratch.data(), packet.wire_size()});
332  }
333  sending_file->mark_completed(symbols, dispatched);
334 
335  // Stats: count the packet, its bytes, the symbol breakdown, and
336  // (if it was a TOI=0 FDT) bump the FDT-emission counter. Symbol
337  // classification: for any file with a non-zero
338  // max_source_block_length, ESI < K is "source", ESI >= K is
339  // "repair". CompactNoCode never has repair (encoder targets
340  // K source symbols only); the breakdown reduces to all-source.
341  if (dispatched) {
342  _stats.packets_emitted.fetch_add(1, std::memory_order_relaxed);
343  _stats.bytes_emitted.fetch_add(packet.wire_size(),
344  std::memory_order_relaxed);
345  const std::uint32_t k_block =
346  sending_file->meta().fec_oti.max_source_block_length;
347  std::uint64_t src = 0, rep = 0;
348  for (const auto& sym : symbols) {
349  if (k_block > 0 && sym.id() >= k_block) ++rep;
350  else ++src;
351  }
352  _stats.source_symbols_emitted.fetch_add(src, std::memory_order_relaxed);
353  _stats.repair_symbols_emitted.fetch_add(rep, std::memory_order_relaxed);
354  if (sending_file->meta().toi == 0) {
355  _stats.fdt_packets_emitted.fetch_add(1, std::memory_order_relaxed);
356  }
357  } else {
358  _stats.packets_failed.fetch_add(1, std::memory_order_relaxed);
359  }
360 
361  // Rate limiting: schedule the next allowed send. The callback's
362  // wall-time cost is paid out of this budget — i.e. if dispatch
363  // took 1 ms, the next deadline is 1 ms closer.
364  if (_rate_limit_kbps > 0 && dispatched) {
365  const std::uint64_t bits = static_cast<std::uint64_t>(packet.wire_size()) * 8ULL;
366  const std::uint64_t ns =
367  (bits * 1'000'000ULL + _rate_limit_kbps - 1ULL) / _rate_limit_kbps;
368  _next_send_due = Clock::now() + std::chrono::nanoseconds{ns};
369  } else {
370  _next_send_due = Clock::time_point::min();
371  }
372 
373  if (sending_file->complete()) {
374  completed_toi = sending_file->meta().toi;
375  // For non-FDT files, we want to invoke the completion callback
376  // outside the lock. Snapshot it here.
377  if (completed_toi != 0) {
378  dispatch_cb = _completion_cb;
379  completed_file = _files[completed_toi];
380  _stats.files_transmitted.fetch_add(1, std::memory_order_relaxed);
381  }
382  file_transmitted_locked(completed_toi);
383  }
384  }
385 
386  if (dispatch_cb && completed_toi != 0) {
387  dispatch_cb(completed_toi);
388  }
389  return true;
390 }

392 std::size_t Encoder::flush(std::size_t max_packets) {
393  std::size_t sent = 0;
