35 // the subset they want to surface to operators.

40  std::uint64_t packets_failed = 0;
41  std::uint64_t bytes_emitted = 0;

43  // Encoding-symbol breakdown across all emitted packets. For
44  // CompactNoCode (no FEC repair) every symbol is a source symbol.
45  // For Raptor, ESI < K is source, ESI >= K is repair (computed
46  // against the file's FDT-declared max_source_block_length, which
47  // matches K for non-trailing blocks; the trailing block of a

63 // FLUTE/ALC transmit-side encoder.
64 //
65 // The encoder builds an ALC packet stream from objects queued via
66 // send(). The CALLER is responsible for transport — each generated
67 // packet is handed back via the PacketCallback registered at
68 // construction. There are no sockets, no event loop, and no internal
69 // threads: pump packets out by calling send_next_packet() (or flush())
70 // from whatever loop already drives your application.
71 //
72 // A typical application either:
73 // - calls flush() until it returns 0, then sleeps until
74 // next_send_due() (rate-limited path), or
75 // - polls send_next_packet() in a tight loop until done (when
76 // rate_limit_kbps == 0).
77 //
78 // Thread-safety: send() and the pump methods are mutex-guarded; it is
79 // safe to enqueue from a different thread than the one driving the
80 // pump. PacketCallback is invoked from whatever thread calls the pump.
81 class Encoder {
82  public:
83  using PacketCallback = std::function<bool(std::span<const std::uint8_t>)>;
84  using CompletionCallback = std::function<void(std::uint32_t toi)>;
85  using Clock = std::chrono::steady_clock;
