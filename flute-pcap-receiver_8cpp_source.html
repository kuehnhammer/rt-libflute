91  spdlog::set_level(static_cast<spdlog::level::level_enum>(args.log_level));
92  spdlog::set_pattern("[%H:%M:%S.%f] [%^%l%$] %v");

94  // Open the pcap file. Build a BPF filter so libpcap drops anything
95  // that isn't UDP for the requested target/port — saves us walking
96  // every frame in C++.
97  char errbuf[PCAP_ERRBUF_SIZE];
98  pcap_t* cap = pcap_open_offline(args.capture_file, errbuf);
99  if (cap == nullptr) {
100  spdlog::error("pcap_open_offline({}): {}", args.capture_file, errbuf);
101  return 1;
102  }
103 
104  std::string filter = "udp dst port " + std::to_string(args.mcast_port) +
105  " and dst host " + args.mcast_target;
106  bpf_program bpf;
107  if (pcap_compile(cap, &bpf, filter.c_str(), 1, PCAP_NETMASK_UNKNOWN) < 0) {
108  spdlog::error("pcap_compile({}): {}", filter, pcap_geterr(cap));
109  pcap_close(cap);
110  return 1;
111  }
112  if (pcap_setfilter(cap, &bpf) < 0) {
113  spdlog::error("pcap_setfilter: {}", pcap_geterr(cap));
114  pcap_freecode(&bpf);
115  pcap_close(cap);
116  return 1;
117  }
118  pcap_freecode(&bpf);
119 
120  LibFlute::Decoder decoder(args.tsi);
121  decoder.register_completion_callback(
122  [&](std::shared_ptr<LibFlute::File> file) {
123  spdlog::info("received {} (TOI {}, {} bytes)",
124  file->meta().content_location, file->meta().toi,
125  file->length());
126  std::string outpath;
127  const char* slash = std::strrchr(file->meta().content_location.c_str(), '/');
128  const char* fname = slash ? slash + 1 : file->meta().content_location.c_str();
129  if (args.download_dir) {
130  outpath = std::string(args.download_dir) + "/" + fname;
131  } else {
132  outpath = "flute_download_" + std::to_string(file->meta().toi) +
133  "-" + fname;
134  }
135  if (FILE* fd = std::fopen(outpath.c_str(), "wb"); fd != nullptr) {
136  std::fwrite(file->buffer(), 1, file->length(), fd);
137  std::fclose(fd);
138  } else {
139  spdlog::error("open({}) for write: {}", outpath, std::strerror(errno));
140  }
141  });
142 
143  // Walk the captured frames. Assume Ethernet → IP v4 → UDP layout
144  // (matches libflute's wire on real Ethernet links). If the pcap was
145  // taken on a loopback interface the link type is DLT_NULL with a
146  // 4-byte AF_* prefix; handle that branch too.
147  pcap_pkthdr* hdr;
148  const u_char* pkt;
149  const int link_type = pcap_datalink(cap);
150  while (pcap_next_ex(cap, &hdr, &pkt) == 1) {
151  std::size_t link_offset = 0;
152  if (link_type == DLT_EN10MB) {
153  link_offset = 14; // Ethernet
154  } else if (link_type == DLT_NULL || link_type == DLT_LOOP) {
155  link_offset = 4; // BSD loopback header
156  } else if (link_type == DLT_RAW) {
157  link_offset = 0;
158  } else {
159  spdlog::warn("unsupported pcap link type {}", link_type);
160  continue;
161  }
162  if (hdr->caplen < link_offset + sizeof(iphdr) + sizeof(udphdr)) continue;
163 
164  const auto* ip = reinterpret_cast<const iphdr*>(pkt + link_offset);
165  const std::size_t ip_hl = static_cast<std::size_t>(ip->ihl) * 4U;
166  if (ip->version != 4 || ip->protocol != IPPROTO_UDP) continue;
167  if (hdr->caplen < link_offset + ip_hl + sizeof(udphdr)) continue;
168 
169  const auto* udp =
170  reinterpret_cast<const udphdr*>(pkt + link_offset + ip_hl);
171  const std::size_t udp_payload_offset = link_offset + ip_hl + sizeof(udphdr);
172  const std::size_t udp_payload_len =
173  static_cast<std::size_t>(ntohs(udp->len)) - sizeof(udphdr);
174  if (hdr->caplen < udp_payload_offset + udp_payload_len) continue;
175 
176  decoder.feed_packet({pkt + udp_payload_offset, udp_payload_len});
177  }
178 
179  pcap_close(cap);
180  return 0;
181 }
