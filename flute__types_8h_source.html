73  // SourceBlock holds the symbols for one RFC 5052 source block.
74  // ESIs run densely from 0..symbols.size()-1, so a vector indexed by
75  // ESI is the natural fit — std::map adds per-symbol heap allocation
76  // and O(log K) lookup with no upside (the keys are dense integers).
77  // For Raptor, slots [0, K) hold source symbols and [K, target_K)
78  // hold repair symbols; for CompactNoCode there are no repair slots.
79  // emit_cursor is a per-block hint used by File::get_next_symbols to
80  // avoid O(K²) scans when emitting all symbols of a block one packet
81  // at a time.
82  //
83  // The Raptor encoder owns its per-block symbol-data scratch inside
84  // RaptorFEC itself (so SourceBlock stays cache-friendly for
85  // CompactNoCode files which can have thousands of blocks).
86  // Symbol::data references either (a) the user file buffer
87  // (CompactNoCode encoder + receiver side, regardless of FEC), or
88  // (b) RaptorFEC's per-block scratch (Raptor encoder side).
89  // completed_symbol_count is maintained by File::put_symbol /
90  // File::mark_completed: incremented once when a Symbol.complete
91  // bit transitions false→true. The FEC layer uses it to answer
92  // "is this block done?" in O(1) instead of std::all_of-ing the
93  // symbols vector after every packet (which was O(K²) per block
94  // on the encoder side at K=9200).
95  struct SourceBlock {
96  uint32_t id = 0;
97  bool complete = false;
98  std::vector<Symbol> symbols;
99  uint32_t emit_cursor = 0;
100  uint32_t completed_symbol_count = 0;
101  };

103  // FEC Object Transmission Information. Mirrors the FEC-OTI-* attribute
104  // set TS 26.346 §7.2.10.1 mandates on every FDT-Instance / File. Any
105  // field left at its sentinel value (0 / empty) is treated as "absent
106  // in the FDT" — the serialiser omits the attribute, and the FEC
107  // transformer fills in a scheme default and writes the resolved value
108  // back into this struct so the FDT round-trip captures it.
109  struct FecOti {
110  FecScheme encoding_id;
111  uint64_t transfer_length;
112  uint32_t encoding_symbol_length;
113  uint32_t max_source_block_length;
114  std::string scheme_specific_info;
115  // FEC-OTI-FEC-Instance-ID. Per RFC 5052 §4, only meaningful for
116  // Under-Specified FEC schemes (Encoding-ID 128–255). The schemes
117  // libflute itself emits — CompactNoCode (0), R10 (1), RaptorQ (6),
118  // RS GF(2^8) (5) — are all Fully-Specified and leave this at 0;
119  // the field exists for FDT round-trip fidelity on the receive
120  // side and forward-compatibility with under-specified schemes.
121  uint8_t instance_id = 0;
122  // FEC-OTI-Max-Number-of-Encoding-Symbols. Upper bound on the ESI
123  // a receiver may see for any source block (= K + max repair).
124  // Sentinel 0 means "let the FEC transformer derive it from the
125  // redundancy level" and the FDT serialiser omits the attribute.
126  uint32_t max_number_of_encoding_symbols = 0;
127  };

129  // Per-file transmission config the caller hands to Encoder::send().
130  // Mirrors the SDP FEC parametrisation surface TS 26.346 actually
131  // exposes for download delivery: §7.3.2.8 (Encoding-ID + optional
132  // Instance-ID via `a=FEC-declaration:`) and §7.3.2.11 (per-file
133  // `a=FEC-redundancy-level:` percent). The remaining FEC OTI fields
134  // (T, K, Z, N, Al, max-encoding-symbols) are libflute's job — they
135  // flow into the FDT XML at runtime per the partitioner, and the
136  // caller has no way to express them in xMB/SDP anyway.
137  //
138  // `sub_block_size_target` is the only field with no SDP backing —
139  // it's a deployment-time knob (TS 26.346 §B.3.4.1 normative 256 KB
140  // for R10, deployment-tuned for RaptorQ per RFC 6330 §4.3) — and
141  // sits here because it's a per-file partitioner input.
142  //
143  // Implicitly constructible from FecScheme so call sites that only
144  // need to pick a scheme can still write
145  // `encoder.send(..., FecScheme::Raptor)` unchanged.
146  struct FileTransmissionConfig {
147  // SDP §7.3.2.8 Encoding-ID. Maps to the wire-level FEC scheme;
148  // identical to the value emitted in the FDT FEC-OTI-FEC-Encoding-ID
149  // attribute.
150  FecScheme scheme = FecScheme::CompactNoCode;
151 
152  // SDP §7.3.2.8 instance-id. Only meaningful for under-specified
153  // FEC schemes (RFC 5052 §4); fully-specified schemes
154  // (CompactNoCode, R10, RaptorQ, RS GF(2^8)) leave this at 0.
155  std::uint8_t fec_instance_id = 0;
156 
157  // SDP §7.3.2.11 redundancy-level. Per-file repair overhead as an
158  // integer percent: 15 ⇒ K * 1.15 total symbols on the wire.
159  // nullopt ⇒ scheme default. Round-trips into the FDT's Rel-11
160  // mbms2012:FEC-Redundancy-Level attribute.
161  std::optional<unsigned> fec_redundancy_level;
162 
163  // RFC 5053 §4.2 W / RFC 6330 §4.3 WS. Sub-block / working-memory
164  // size target. Sentinel 0 ⇒ use the scheme default (16 MB).
165  // TS 26.346 §B.3.4.1 mandates 256 KB for R10 file delivery;
166  // RaptorQ leaves it as a deployment knob. Not on the SDP wire —
167  // the integrator picks this per their build/deployment config.
168  std::uint64_t sub_block_size_target = 0;
169 
170  // RFC 6330 §4.3 SS — desired lower bound on sub-symbol size,
171  // expressed as a multiplier on Al. The §4.3 algorithm picks
172  // N ≤ T/(SS·Al), so SS=1 (default) leaves the algorithm free to
173  // pick the largest N the partition allows; SS>1 forces fewer,
174  // larger sub-symbols. RaptorQ-only knob; ignored for R10.
175  std::uint8_t sub_symbol_size_min_multiplier = 1;
176 
177  FileTransmissionConfig() = default;
178 
179  // NOLINTNEXTLINE(google-explicit-constructor) — implicit conversion
180  // from FecScheme keeps existing `encoder.send(..., FecScheme::X)`
181  // call sites compiling.
182  FileTransmissionConfig(FecScheme s) : scheme(s) {}
183  };
