38 std::size_t Decoder::flush_pending_decodes() {
39  // Snapshot the (toi, file) pairs under the lock, then drop the
40  // lock for each file's try_decode_pending — the FEC matrix-solve
41  // can be hundreds of ms and we don't want to hold _files_mutex
42  // across that (feed_packet on another thread should still be able
43  // to land late repair symbols, in principle).
44  std::vector<std::pair<std::uint32_t, std::shared_ptr<File>>> snapshot;
45  {
46  const std::lock_guard<std::mutex> lock(_files_mutex);
47  snapshot.reserve(_files.size());
48  for (auto& [toi, file] : _files) {
49  if (file && !file->complete() && toi != 0) {
50  snapshot.emplace_back(toi, file);
51  }
52  }
53  }
54 
55  std::size_t completed = 0;
56  for (auto& [toi, file] : snapshot) {
57  file->try_decode_pending();
58  if (!file->complete()) {
59  continue;
60  }
61  ++completed;
62  spdlog::debug("flush_pending_decodes: TOI {} completed", toi);
63 
64  // Fire the completion callback under the same protocol the FDT
65  // path uses (snapshot under lock, dispatch outside).
66  CompletionCallback cb_snapshot;
67  std::shared_ptr<File> cb_file;
68  {
69  const std::lock_guard<std::mutex> lock(_files_mutex);
70  auto fit = _files.find(toi);
71  if (fit != _files.end() && _completion_cb) {
72  cb_file = fit->second;
73  cb_snapshot = _completion_cb;
74  _files.erase(fit);
75  }
76  }
77  if (cb_snapshot && cb_file) {
78  _stats.files_completed.fetch_add(1, std::memory_order_relaxed);
79  cb_snapshot(cb_file);
80  }
81  }
82  return completed;
83 }

95 void Decoder::feed_packet(std::span<const std::uint8_t> alc_payload) {
96  // AlcPacket parses the input read-only, but the existing
97  // constructor signature takes a non-const char* (legacy from when
98  // sockets handed it a mutable receive buffer). const_cast is safe.
99  char* data = const_cast<char*>(
100  reinterpret_cast<const char*>(alc_payload.data()));
101  const std::size_t bytes = alc_payload.size();
102 
103  _stats.packets_received.fetch_add(1, std::memory_order_relaxed);
104  _stats.bytes_received.fetch_add(bytes, std::memory_order_relaxed);
105 
106  // File completions are dispatched outside the mutex; collect them
107  // here and invoke the callback after the lock is released so a
108  // callback that re-enters the Decoder cannot deadlock.
109  std::shared_ptr<File> completed_file;
110  CompletionCallback callback_snapshot;
111 
112  try {
113  auto alc = AlcPacket(data, bytes);
114 
115  if (alc.tsi() != _tsi) {
116  spdlog::warn("Discarding packet for unknown TSI {}", alc.tsi());
117  _stats.dropped_wrong_tsi.fetch_add(1, std::memory_order_relaxed);
118  return;
119  }
120 
121  if (bytes < alc.header_length()) {
122  spdlog::warn("Discarding packet: header_length {} exceeds packet size {}",
123  alc.header_length(), bytes);
124  _stats.dropped_truncated.fetch_add(1, std::memory_order_relaxed);
125  return;
126  }
127 
128  {
129  const std::lock_guard<std::mutex> lock(_files_mutex);
130 
131  // RFC 6726 §3.3 FDT-Instance routing.
132  //
133  // (a) older than committed _fdt OR older than the in-flight
134  // TOI=0 File's instance — drop. Replays / out-of-order.
135  // (b) matches the in-flight File's instance — feed bytes via
136  // the lower path (existing logic).
137  // (c) newer than everything we know — replace any in-flight
138  // File (sized for the OLD instance's transfer_length)
139  // with a fresh one sized for THIS packet's EXT_FTI.
140  // Without this, a v=N+1 packet arriving mid-receive of
141  // v=N would feed bytes into the v=N File and corrupt
142  // both. (RFC 6726 §3.3 + RFC 1982 circular comparison.)
143  if (alc.toi() == 0) {
144  const std::uint32_t new_id = alc.fdt_instance_id();
145  auto inflight_it = _files.find(0);
146  const bool have_inflight = inflight_it != _files.end();
147  const std::uint32_t inflight_id =
148  have_inflight ? inflight_it->second->fdt_instance_id() : 0;
149 
150  const bool stale_vs_committed =
151  _fdt && !FileDeliveryTable::IsNewerInstanceId(
152  new_id, _fdt->instance_id());
153  const bool stale_vs_inflight =
154  have_inflight && new_id != inflight_id &&
155  !FileDeliveryTable::IsNewerInstanceId(new_id, inflight_id);
156 
157  if (stale_vs_committed || stale_vs_inflight) {
158  spdlog::debug("Discarding stale FDT packet (instance_id {})",
159  new_id);
160  _stats.dropped_stale_fdt.fetch_add(1, std::memory_order_relaxed);
161  return;
162  }
163 
164  if (have_inflight && new_id != inflight_id) {
165  spdlog::debug("FDT packet supersedes in-flight {} -> {}",
166  inflight_id, new_id);
167  _files.erase(0);
168  }
169 
170  if (_files.find(0) == _files.end()) {
171  FileDeliveryTable::FileEntry fe{};
172  fe.content_length = alc.fec_oti().transfer_length;
173  fe.fec_oti = alc.fec_oti();
174  auto file = std::make_shared<File>(fe);
175  file->set_fdt_instance_id(new_id);
176  _files.emplace(0, file);
177  }
178  }
179 
180  const auto file_it = _files.find(alc.toi());
181  const bool file_present = file_it != _files.end();
182  const bool file_already_complete =
183  file_present && file_it->second->complete();
184  if (file_present && !file_already_complete) {
185  auto encoding_symbols = EncodingSymbol::from_payload(
186  data + alc.header_length(),
187  bytes - alc.header_length(),
188  _files[alc.toi()]->fec_oti(),
189  alc.content_encoding());
190 
191  // Symbol-class breakdown. ESI < K (file's
192  // max_source_block_length) is "source", ESI >= K is "repair".
193  // CompactNoCode never sends repair so the receiver naturally
194  // counts every symbol as source.
195  const std::uint32_t k_block =
196  _files[alc.toi()]->fec_oti().max_source_block_length;
197  std::uint64_t src = 0, rep = 0;
198  for (const auto& symbol : encoding_symbols) {
199  spdlog::debug("received TOI {} SBN {} ID {}",
200  alc.toi(), symbol.source_block_number(), symbol.id());
201  if (k_block > 0 && symbol.id() >= k_block) ++rep;
202  else ++src;
203  _files[alc.toi()]->put_symbol(symbol);
204  }
205  _stats.source_symbols_received.fetch_add(src, std::memory_order_relaxed);
206  _stats.repair_symbols_received.fetch_add(rep, std::memory_order_relaxed);
207 
208  File* file = _files[alc.toi()].get();
209  if (_files[alc.toi()]->complete()) {
210  for (auto it = _files.begin(); it != _files.end();) {
211  if (it->second.get() != file &&
212  it->second->meta().content_location ==
213  file->meta().content_location) {
214  spdlog::debug("Replacing file with TOI {}", it->first);
215  it = _files.erase(it);
216  } else {
217  ++it;
218  }
219  }
220 
221  spdlog::debug("File with TOI {} completed", alc.toi());
222  if (alc.toi() != 0) {
223  _stats.files_completed.fetch_add(1, std::memory_order_relaxed);
224  if (_completion_cb) {
225  // Snapshot under the lock; dispatch after release.
226  completed_file = _files[alc.toi()];
227  callback_snapshot = _completion_cb;
228  _files.erase(alc.toi());
229  }
230  }
231 
232  if (alc.toi() == 0) { // parse complete FDT
233  auto candidate = std::make_unique<FileDeliveryTable>(
234  alc.fdt_instance_id(), _files[alc.toi()]->buffer(),
235  _files[alc.toi()]->length(),
236  _fec_dec_worker_threads);
237  _files.erase(alc.toi());
238 
239  // RFC 6726 §3.3: an FDT-Instance MUST NOT be relied upon
240  // after its Expires time. The current FDT (if any) stays
241  // in force; the expired candidate is dropped.
242  if (candidate->is_expired(_now())) {
243  spdlog::debug("Discarding expired FDT-Instance ID {}",
244  alc.fdt_instance_id());
245  _stats.fdts_rejected_expired.fetch_add(1,
246  std::memory_order_relaxed);
247  } else {
248  // Capture old-FDT TOI set BEFORE replacing _fdt; TOIs in
249  // the old set but not the new are "abandoned" — the
250  // sender has stopped emitting symbols for them, so the
251  // FEC layer should attempt a final decode pass with
252  // whatever symbols arrived. RFC 6726 §3.3 ¶5 says the
253  // file itself stays in file_list; only the decode
254  // trigger fires.
255  std::vector<std::uint32_t> old_tois;
256  if (_fdt) {
257  for (const auto& fe : _fdt->file_entries()) {
258  old_tois.push_back(fe.toi);
259  }
260  }
261 
262  _fdt = std::move(candidate);
263  _stats.fdts_accepted.fetch_add(1, std::memory_order_relaxed);
264  for (const auto& file_entry : _fdt->file_entries()) {
265  if (_files.find(file_entry.toi) == _files.end()) {
266  spdlog::debug("Starting reception for TOI {}: {} ({})",
267  file_entry.toi, file_entry.content_location,
268  file_entry.content_type);
269  _files.emplace(file_entry.toi,
270  std::make_shared<File>(file_entry));
271  }
272  }
273 
274  // Build the abandoned-TOI set and trigger a decode pass
275  // on each one.
276  std::vector<std::uint32_t> new_tois;
277  for (const auto& fe : _fdt->file_entries()) {
278  new_tois.push_back(fe.toi);
279  }
280  for (auto t : old_tois) {
281  bool still_listed = false;
282  for (auto n : new_tois) {
283  if (n == t) { still_listed = true; break; }
284  }
285  if (still_listed) continue;
286  auto fit = _files.find(t);
287  if (fit == _files.end()) continue;
288  auto& file = fit->second;
289  if (file->complete()) continue;
290  spdlog::debug("FDT no longer lists TOI {}; "
291  "triggering end-of-transmission decode",
292  t);
293  file->try_decode_pending();
294  if (file->complete()) {
295  _stats.files_completed.fetch_add(
296  1, std::memory_order_relaxed);
297  if (_completion_cb && completed_file == nullptr) {
298  completed_file = file;
299  callback_snapshot = _completion_cb;
300  _files.erase(fit);
301  }
302  }
303  }
304  }
305  }
306  }
307  } else if (file_already_complete) {
308  // Bonus symbol(s) arriving after the file (typically a
309  // Raptor source block) was already successfully decoded.
310  // Not a drop — just unused redundancy. Silent skip; not
311  // worth a counter because it's a normal Raptor scenario.
312  spdlog::trace("Ignoring symbol for already-complete TOI {}",
313  alc.toi());
314  } else {
315  // No File for this TOI — receiver hasn't seen the FDT yet,
316  // or the TOI is for a different session entirely.
317  spdlog::trace("Discarding packet for unknown TOI {}", alc.toi());
318  _stats.dropped_no_matching_file.fetch_add(1,
319  std::memory_order_relaxed);
320  }
321  }
322  } catch (const std::exception& ex) {
323  spdlog::warn("Failed to decode ALC/FLUTE packet: {}", ex.what());
324  _stats.dropped_malformed.fetch_add(1, std::memory_order_relaxed);
325  return;
326  }
327 
328  if (callback_snapshot && completed_file) {
329  callback_snapshot(std::move(completed_file));
330  }
331 }

343 void Decoder::remove_expired_files(unsigned max_age_seconds) {
344  const std::lock_guard<std::mutex> lock(_files_mutex);
345  for (auto it = _files.cbegin(); it != _files.cend();) {
346  auto age = std::time(nullptr) - it->second->received_at();
347  if (it->second->meta().content_location != "bootstrap.multipart" &&
348  age > max_age_seconds) {
349  // Incomplete-at-eviction = the FEC layer (or the lossless
350  // accumulator for CompactNoCode) couldn't recover the file
351  // before its TTL expired. This is the closest signal we have
352  // to "FEC repair gave up".
353  if (it->first != 0 && !it->second->complete()) {
354  _stats.files_discarded_incomplete.fetch_add(
355  1, std::memory_order_relaxed);
356  _stats.bytes_discarded_incomplete.fetch_add(
357  it->second->meta().fec_oti.transfer_length,
358  std::memory_order_relaxed);
359  }
360  it = _files.erase(it);
361  } else {
362  ++it;
363  }
364  }
