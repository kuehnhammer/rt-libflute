35 // Decoder operational counters. Snapshot returned by
36 // Decoder::stats(). Lifetime-cumulative; values monotonically
37 // non-decreasing. The library exposes the full counter set; consumers
38 // pick what they want to surface to operators.
39 struct DecoderStats {
40  // Packets observed at feed_packet().
41  std::uint64_t packets_received = 0;
42  std::uint64_t bytes_received = 0;
43 
44  // Per-packet drop reasons. packets_dropped_total() sums them.
45  // (Expired-FDT rejection is NOT a packet drop — by the time the
46  // expiry check fires the bytes have already been accumulated; see
47  // fdts_rejected_expired below for that counter.)
48  std::uint64_t dropped_wrong_tsi = 0;
49  std::uint64_t dropped_malformed = 0; // parser threw
50  std::uint64_t dropped_truncated = 0; // header_length > packet bytes
51  std::uint64_t dropped_stale_fdt = 0; // TOI=0 packet for older FDT
52  std::uint64_t dropped_no_matching_file = 0; // TOI != 0, no FDT entry yet
53  // OR file already complete
54 
55  std::uint64_t packets_dropped_total() const {
56  return dropped_wrong_tsi + dropped_malformed + dropped_truncated +
57  dropped_stale_fdt + dropped_no_matching_file;
58  }
59 
60  // Encoding-symbol breakdown. For CompactNoCode every symbol is a
61  // source symbol. For Raptor, ESI < K (file's max_source_block_length)
62  // is "source", ESI >= K is "repair".
63  std::uint64_t source_symbols_received = 0;
64  std::uint64_t repair_symbols_received = 0;
65 
66  // File-level counters.
67  std::uint64_t files_completed = 0;
68  // Incomplete files evicted via remove_expired_files() or
69  // remove_file_with_content_location() — i.e. reception was started
70  // (the FDT announced this TOI) but the receiver never accumulated
71  // enough symbols to finish, even with FEC repair if applicable.
72  std::uint64_t files_discarded_incomplete = 0;
73  // Sum of declared transfer_length over all files counted in
74  // files_discarded_incomplete. Useful for "what % of announced
75  // payload couldn't be recovered" dashboards.
76  std::uint64_t bytes_discarded_incomplete = 0;
77 
78  std::uint64_t fdts_accepted = 0;
79  std::uint64_t fdts_rejected_expired = 0;
80  std::uint64_t fdts_rejected_stale = 0;
81 };

83 // FLUTE/ALC receive-side decoder.
84 //
85 // The decoder accepts already-extracted ALC packet payloads via
86 // feed_packet(); the CALLER is responsible for transport (recvfrom on
87 // a UDP socket, RLC SDU reassembly out of the broadcast pipeline,
88 // pcap replay, …). There are no sockets, no event loop, and no
89 // internal threads.
90 //
91 // Lifecycle:
92 // 1. Construct with the joined session's TSI.
93 // 2. Register a completion callback (optional but typical).
94 // 3. Call feed_packet() for every incoming ALC payload.
95 // 4. The callback fires once per file when its source symbols have
96 // all been received (FEC-repaired, if applicable).
97 //
98 // Thread-safety: feed_packet() and the inspection methods are
99 // mutex-guarded. The completion callback is invoked from the same
100 // thread that called feed_packet() at the moment of completion;
101 // re-entrant calls into the Decoder from inside the callback are
102 // safe (the lock is released before dispatch).
